(function(){ const role = localStorage.getItem('role'); if(role !== 'user'){ alert('Accès utilisateur requis. Retour au login.'); window.location.href = 'index.html'; return; } let payments = JSON.parse(localStorage.getItem("payments") || "[]"); const app = document.getElementById('app'); app.innerHTML = `

Espace Utilisateur

Bonjour ${localStorage.getItem('email') || 'Utilisateur'}

Historique des paiements

${ payments.length === 0 ? `

Aucun paiement enregistré.

` : payments.map((p, i) => `

Montant : ${p.amount} €

Date : ${p.date}

Carte : ${maskCard(p.cardNumber)}

Expiration : ${p.exp}

`).join('') }

Message du paiement

`; function maskCard(num) { if(!num || num.length < 16) return "Carte invalide"; return num.substring(0,4) + " **** **** " + num.substring(12); } // Logout document.getElementById('btnLogout').addEventListener('click', () => { localStorage.removeItem('role'); localStorage.removeItem('email'); window.location.href = 'index.html'; }); // Modal const modal = document.getElementById("modalMessage"); const modalContent = document.getElementById("modalContent"); const closeBtn = document.getElementById("btnCloseModal"); closeBtn.addEventListener('click', () => { modal.classList.add("hidden"); }); function attachViewMessageButtons() { document.querySelectorAll('.btnViewMessage').forEach(btn => { btn.addEventListener('click', () => { const index = btn.dataset.index; const msg = payments[index].message || "Aucun message"; modalContent.innerHTML = msg; modal.classList.remove("hidden"); modal.classList.add("flex"); }); }); } attachViewMessageButtons(); // Gestion du formulaire "Nouveau paiement" const btnNewPay = document.getElementById('btnNewPay'); const mainFlex = document.getElementById('mainFlexContainer'); const paymentFormHTML = `

Nouveau paiement

`; btnNewPay.addEventListener('click', () => { if (!document.getElementById('newPaymentFormContainer')) { mainFlex.insertAdjacentHTML('beforeend', paymentFormHTML); document.getElementById('newPaymentForm').addEventListener('submit', e => { e.preventDefault(); const amount = document.getElementById('paymentAmount').value; const card = document.getElementById('cardNumber').value; const exp = document.getElementById('cardExp').value; const message = document.getElementById('paymentMessage').value; payments.push({ id: payments.length + 1, amount, cardNumber: card, exp, message, date: new Date().toISOString().split('T')[0] }); localStorage.setItem('payments', JSON.stringify(payments)); updatePaymentList(); }); } }); // Fonction pour rafraîchir l'historique après ajout function updatePaymentList() { const paymentList = document.getElementById('paymentList'); paymentList.innerHTML = payments.map((p, i) => `

Montant : ${p.amount} €

Date : ${p.date}

Carte : ${maskCard(p.cardNumber)}

Expiration : ${p.exp}

`).join(''); attachViewMessageButtons(); } })();
